=== njli-demo-build_week.service ===
[Unit]
Description=njli demo: build_week

[Service]
Type=oneshot
WorkingDirectory=@WS@
Environment=PATH=/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin
Environment=PYTHONUTF8=1
Environment=NJLI_NODE=/opt/homebrew/bin/node
Environment=NJLI_NPX=/opt/homebrew/bin/npx
Environment=NJLI_FFMPEG=/opt/homebrew/bin/ffmpeg
Environment=NJLI_FFPROBE=/opt/homebrew/bin/ffprobe
Environment=NJLI_CLAUDE_BIN=@HOME@/.local/bin/claude
ExecStart=/usr/bin/python3 @PYZ@ run build_week --workspace @WS@ --no-jitter
TimeoutStartSec=11100
=== njli-demo-build_week.timer ===
[Unit]
Description=njli demo: build_week (THU 10:00)

[Timer]
OnCalendar=Thu *-*-* 10:00:00
Persistent=true
RandomizedDelaySec=20min
Unit=njli-demo-build_week.service

[Install]
WantedBy=timers.target
